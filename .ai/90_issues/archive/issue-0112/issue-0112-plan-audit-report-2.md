```bash
check_a "$clause_a" && ok '테스트 관점: SKILL 정상' || ng '테스트 관점: SKILL 정상'
check_b "$clause_b" && ok '테스트 관점: spec 정상' || ng '테스트 관점: spec 정상'
# 다음 두 assertion을 제거해도 원문 완료 기준이 통과했다.
if check_a ''; then ng '테스트 관점: SKILL 삭제 반례'; else ok '테스트 관점: SKILL 삭제 반례'; fi
if check_b ''; then ng '테스트 관점: spec 삭제 반례'; else ok '테스트 관점: spec 삭제 반례'; fi
```
